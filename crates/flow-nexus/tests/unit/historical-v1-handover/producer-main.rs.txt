mod historical;
use historical::{StoredHandoverConfiguration,DescribesHandoverConfiguration};
use sema_engine::{Engine,EngineOpen,SchemaVersion,Assertion};
use meta_signal_flow::{HandoverPoliciesConfiguration,HandoverArchiveConfiguration,HandoverCountConfiguration,HandoverPolicy};
fn main() {
 let directory=std::path::PathBuf::from(std::env::args().nth(1).expect("output directory"));
 std::fs::create_dir_all(&directory).unwrap();
 for (name, policies, archive, count) in [
  ("unconfigured",HandoverPoliciesConfiguration::Unconfigured,HandoverArchiveConfiguration::Unconfigured,HandoverCountConfiguration::Unconfigured),
  ("empty-policy",HandoverPoliciesConfiguration::Configured(vec![]),HandoverArchiveConfiguration::Configured("/historical/empty policy/archive".into()),HandoverCountConfiguration::Configured(7)),
  ("nonempty-policy",HandoverPoliciesConfiguration::Configured(vec![HandoverPolicy {model_name:"historicalModelOne".into(),warning_share_left:17,reminder_text:"Historical reminder one: preserve nested text.".into(),every_minutes:3},HandoverPolicy {model_name:"historicalModelTwo".into(),warning_share_left:29,reminder_text:"Historical reminder two: distinct bytes.".into(),every_minutes:11}]),HandoverArchiveConfiguration::Configured("/historical/nonempty policy/archive".into()),HandoverCountConfiguration::Configured(13))
 ] {
  let path=directory.join(format!("{name}.sema"));
  let mut engine=Engine::open(EngineOpen::new(&path,SchemaVersion::new(1))).unwrap();
  let table=engine.register_table(StoredHandoverConfiguration::descriptor()).unwrap();
  engine.assert(Assertion::new(table,StoredHandoverConfiguration{policies,archive,count})).unwrap();
  drop(engine);
  println!("{}",path.display());
 }
}
